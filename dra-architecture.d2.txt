# ── Flows ────────────────────────────────────────────────────────────────────

# User submits workload
User -> Workload.Owning Resource: "kubectl apply Deployment"

# Admission converts legacy claims
Workload.Pods.POD -> MutatingAdmissionPolicy.Policy: "ResourceClaim created (legacy opaque config)"

# Scheduling
Workload.Pods.POD -> Kubernetes Scheduler.Structured Parameters: "pending pod with claim"
DRA Node Driver.ResourceSlice CXI -> Kubernetes Scheduler.Structured Parameters: "per-device consumable capacity"

# Binding gate
Kubernetes Scheduler.Structured Parameters -> DRA Node Driver.ClaimBinder: "claim allocated to this node"
DRA Node Driver.ClaimBinder -> Kubernetes Scheduler.Structured Parameters: "Ready=True (bind) / Unschedulable (retry)"

# Namespace VNI
DRA Node Driver.Namespace VNI Allocator -> Node Resources.Namespace: "patch annotation"

# Node preparation
Kubernetes Scheduler.Structured Parameters -> DRA Node Driver.PrepareResourceClaims: "pod scheduled to node"
Node Resources.Namespace -> DRA Node Driver.PrepareResourceClaims: "read VNI from annotation"
DRA Node Driver.PrepareResourceClaims -> Container.Slingshot CNI: "hint file (before RunPodSandbox)"
Node Resources.HSN Interface -> Container.Slingshot CNI: "IPVLAN parent"

# Container lifecycle (NRI)
DRA Node Driver.NRI Plugin -> Node Resources.CXI Service: "CreateService on RunPodSandbox"
Node Resources.CXI Device -> Container.Pod Container: "device node via CDI"
Node Resources.CXI Service -> Container.Pod Container: "restricts access to (netns, VNI)"
